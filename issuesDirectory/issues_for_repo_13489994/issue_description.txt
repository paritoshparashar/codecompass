When using iter_bytes with chunk_size any data that is not filling up a full chunk gets lost.

I'm porting a Python library from requests to httpx. I'm replacing a requests.get(stream=True... with httpx.stream("GET"....

In the requests version I'm using:

                        for chunk in resp.iter_content(chunk_size=8 * 1024):
                            buf.write(chunk)
which I replaced with:

                        for chunk in resp.iter_bytes(chunk_size=8 * 1024):
                            buf.write(chunk)
The endpoint returns 10920 bytes, and the requests block reads them all. The httpx block only reads 8192 bytes.

I changed the chunk_size to 1024 and the httpx block only reads 10240 bytes.

If the chunks are not full, the leftover data is lost.

Here is a demo to show that when using a chunk_size, data gets lost. Included below, you will find the server code, the client code, example runs proving the point, as well as a client implementation using requests, which works fine.

Server
# pip install flask
# python server.py
# curl --trace-ascii /dev/stdin http://127.0.0.1:5000/stream

import time
import threading
import socket
from flask import Flask, Response
from werkzeug.serving import WSGIRequestHandler

app = Flask(__name__)

CHUNK_DATA = "Lorem ipsum dolor sit amet, consectetur adipiscing elit. Phasellus dignissim sem ut erat sagittis interdum."
N_CHUNKS = 3
CHUNK_SIZE = 48
CHUNK_SLEEP = 2  # seconds
RESPONSE_TIMEOUT = 5  # seconds


class TimeoutWSGIRequestHandler(WSGIRequestHandler):
    """Custom request handler that allows access to the socket"""

    def handle_one_request(self):
        """Override to enforce the connection time length"""
        timer = threading.Timer(
            RESPONSE_TIMEOUT, terminate_connection, args=[self.connection]
        )
        timer.daemon = True
        timer.start()
        return super().handle_one_request()


def terminate_connection(connection):
    """Enfore connection time length by terminating the connection"""
    print(f"Connection timeout {RESPONSE_TIMEOUT} seconds. Closing connection.")
    connection.shutdown(socket.SHUT_RDWR)
    connection.close()


def generator():
    for i in range(N_CHUNKS):
        print(f"Sleep {CHUNK_SLEEP} and send {CHUNK_SIZE} bytes")
        time.sleep(CHUNK_SLEEP)  # Simulate delay for chunk generation
        yield CHUNK_DATA[i * CHUNK_SIZE : (i + 1) * CHUNK_SIZE]


@app.route("/stream")
def stream_random_data():
    return Response(generator(), status=200)


if __name__ == "__main__":
    # Run with our custom request handler
    app.run(
        debug=True, port=5000, threaded=True, request_handler=TimeoutWSGIRequestHandler
    )
Client
# pip install httpx
# python client.py
# python client.py 64

import httpx
import sys


def stream_data(url: str):
    bytes_count = 0
    with httpx.Client() as client:
        try:
            with client.stream("GET", url) as response:
                for chunk in response.iter_bytes(
                    chunk_size=int(sys.argv[1]) if len(sys.argv) > 1 else None
                ):
                    bytes_count += len(chunk)
                    buf = chunk.decode("utf-8")
                    print(f"Buffer size: {len(buf)}")
                    print(f"Buffer content: {buf}")
        except httpx.RemoteProtocolError:
            print("Connection closed by the server.")
    print(f"Total bytes received: {bytes_count}")


def main():
    url = "http://localhost:5000/stream"
    stream_data(url)


if __name__ == "__main__":
    main()

Client using requests
# pip install requests
# python client-requests.py
# python client-requests.py 32
# python client-requests.py 64

import requests
import sys


def stream_data(url: str):
    bytes_count = 0
    try:
        with requests.get(url, stream=True) as response:
            chunk_size = int(sys.argv[1]) if len(sys.argv) > 1 else None
            for chunk in response.iter_content(
                chunk_size=chunk_size, decode_unicode=True
            ):
                if chunk:  # filter out keep-alive new chunks
                    bytes_count += len(chunk.encode("utf-8"))  # count actual bytes
                    print(f"Buffer size: {len(chunk)}")
                    print(f"Buffer content: {chunk}")
    except requests.exceptions.RequestException as e:
        print(f"Connection error: {e}")
    print(f"Total bytes received: {bytes_count}")


def main():
    url = "http://localhost:5000/stream"
    stream_data(url)


if __name__ == "__main__":
    main()
